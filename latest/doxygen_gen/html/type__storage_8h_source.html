94template <typename T>
95const inja::json& TypeStorage::getOrCreate(const T& type)
96{
97 const std::string qualName(type.getQualifiedName());
98 auto itr = map_.find(qualName);
99 if (itr != map_.end())
100 {
101 return itr->second;
102 }
103
104 auto [insertItr, inserted] = map_.insert({qualName, toJson(type)});
105 SEN_ENSURE(inserted);
106 return insertItr->second;
107}
108
109const inja::json& TypeStorage::get(const sen::CustomType& type) const
110{
111 const std::string qualName(type.getQualifiedName());
112 auto itr = map_.find(qualName);
113 if (itr == map_.end())
114 {
115 std::string err;
116 err.append("type ");
117 err.append(qualName);
118 err.append(" not found in the computed info");
119 sen::throwRuntimeError(err);
120 }
121
122 return itr->second;
123}
