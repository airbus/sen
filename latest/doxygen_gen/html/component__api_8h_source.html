432template <typename T, typename Bus>
433inline std::shared_ptr<Subscription<T>> KernelApi::selectAllFrom(const Bus& bus)
434{
435 auto sub = std::make_shared<Subscription<T>>();
436 sub->attachTo(getSource(bus), Interest::make(buildQuery<T>(bus), getTypes()), true);
437 return sub;
438}
439
440template <typename T, typename Bus>
441inline std::shared_ptr<Subscription<T>> KernelApi::selectAllFrom(const Bus& bus,
442 typename ObjectList<T>::Callback onAdded,
443 typename ObjectList<T>::Callback onRemoved)
444{
445 auto sub = std::make_shared<Subscription<T>>();
446 // Install callbacks before subscribing so they fire for objects already present.
447 if (onAdded)
448 {
449 std::ignore = sub->list.onAdded(std::move(onAdded));
450 }
451 if (onRemoved)
452 {
453 std::ignore = sub->list.onRemoved(std::move(onRemoved));
454 }
455 sub->attachTo(getSource(bus), Interest::make(buildQuery<T>(bus), getTypes()), true);
456 return sub;
457}

460inline std::shared_ptr<Subscription<T>> KernelApi::selectFrom(const Bus& bus,
461 const std::string& query,
462 typename sen::ObjectList<T>::Callback onAdded,
463 typename sen::ObjectList<T>::Callback onRemoved)

486template <typename T>
487[[nodiscard]] inline std::string_view queryTypeName()
488{
489 if constexpr (std::is_same_v<T, Object>)
490 {
491 return "*";
492 }
493 else
494 {
495 static_assert(std::is_same_v<decltype(T::meta()), ::sen::ConstTypeHandle<::sen::ClassType>>,
496 "a subscription type must be a Sen object class");
497 return T::meta()->getQualifiedName();
498 }
499}

503template <typename T>
504inline std::string KernelApi::buildQuery(const BusAddress& address) const
505{
506 std::string query = "SELECT ";
507 query.append(impl::queryTypeName<T>());
508 query.append(" FROM ");
509 query.append(address.sessionName);
510 query.append(".");
511 query.append(address.busName);
512
513 return query;
514}
515
516template <typename T>
517inline std::string KernelApi::buildQuery(std::string_view bus) const
518{
519 std::string query = "SELECT ";
520 query.append(impl::queryTypeName<T>());
521 query.append(" FROM ");
522 query.append(bus);
523
524 return query;
525}
