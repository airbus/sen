569template <typename T, typename Bus>
570inline std::shared_ptr<Subscription<T>> KernelApi::selectAllFrom(const Bus& bus)
571{
572 auto sub = std::make_shared<Subscription<T>>();
573 sub->attachTo(getSource(bus), Interest::make(buildQuery<T>(bus), getTypes()), true);
574 return sub;
575}
576
577template <typename T, typename Bus>
578inline std::shared_ptr<Subscription<T>> KernelApi::selectAllFrom(const Bus& bus,
579 typename ObjectList<T>::Callback onAdded,
580 typename ObjectList<T>::Callback onRemoved)
581{
582 auto sub = std::make_shared<Subscription<T>>();
583 // Install callbacks before subscribing so they fire for objects already present.
584 if (onAdded)
585 {
586 std::ignore = sub->list.onAdded(std::move(onAdded));
587 }
588 if (onRemoved)
589 {
590 std::ignore = sub->list.onRemoved(std::move(onRemoved));
591 }
592 sub->attachTo(getSource(bus), Interest::make(buildQuery<T>(bus), getTypes()), true);
593 return sub;
594}

623template <typename T>
624[[nodiscard]] inline std::string_view queryTypeName()
625{
626 if constexpr (std::is_same_v<T, Object>)
627 {
628 return "*";
629 }
630 else
631 {
632 static_assert(std::is_same_v<decltype(T::meta()), ::sen::ConstTypeHandle<::sen::ClassType>>,
633 "a subscription type must be a Sen object class");
634 return T::meta()->getQualifiedName();
635 }
636}

640template <typename T>
641inline std::string KernelApi::buildQuery(const BusAddress& address) const
642{
643 std::string query = "SELECT ";
644 query.append(impl::queryTypeName<T>());
645 query.append(" FROM ");
646 query.append(address.sessionName);
647 query.append(".");
648 query.append(address.busName);
649
650 return query;
651}
