31// How far a thing is, in metres.
32quantity<f32, m> Metres;

34// The colours a thing may be painted.
35enum Colour : u8 { red, green, blue }

37// A place, given as two distances.

39
40// A round thing, which is also a place.
41struct Circle: Point { centre : Point, radius : Metres }
42
43// Either a place or a round thing.
44variant Figure { Point, Circle }
45
46// Where something has been.
47sequence<Point, 8> Track;
48
49// Three distances, always three.
50array<Metres, 3> Triple;
51
52// A colour, or nothing.
53optional<Colour> MaybeColour;
54
55// Another name for a distance.
56alias Distance Metres;
57
58// The root of the little hierarchy.
59class Base
60{
61 // How many of it there are.
62 var one : Metres;
63
64 // Asked to settle down.
65 fn settle();
66}
67
68// One step down.
69class Middle: extends Base
70{
71 // Whether it is switched on.
72 var live : bool [writable];
73
74 // Emitted when it stops.
75 event stopped(why : Colour);
76}
77
78// Two steps down, and where most of the members are.
