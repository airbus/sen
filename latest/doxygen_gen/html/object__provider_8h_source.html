241inline ObjectRemoval makeRemoval(const ObjectAddition& discovery)
242{
243 if (std::holds_alternative<ObjectInstanceDiscovery>(discovery))
244 {
245 const auto& instanceDiscovery = std::get<ObjectInstanceDiscovery>(discovery);
246 return {instanceDiscovery.interestId, instanceDiscovery.id, instanceDiscovery.ownerId};
247 }
248
249 const auto& remoteObjectDiscovery = std::get<RemoteObjectDiscovery>(discovery);
250 return {remoteObjectDiscovery.interestId, remoteObjectDiscovery.id, remoteObjectDiscovery.ownerId};
251}
