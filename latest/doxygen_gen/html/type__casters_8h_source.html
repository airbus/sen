24template <>
25struct type_caster<sen::ObjectId>
26{
27public:
28 PYBIND11_TYPE_CASTER(sen::ObjectId, const_name("ObjectId"));
29
30 // Accept only Python ints. PyNumber_Long would silently coerce bools/strings/bytes
31 // (True -> ObjectId(1), "5" -> ObjectId(5)), which is the wrong policy for an id-shaped
32 // strong type.
33 bool load(handle src, bool /*convert*/)
34 {
35 if (PyLong_Check(src.ptr()) == 0)
36 {
37 return false;
38 }
39 // unsigned long is the return type of the CPython API, not a width choice of ours.
40 // NOLINTNEXTLINE(google-runtime-int)
41 const unsigned long raw = PyLong_AsUnsignedLong(src.ptr());
42 // NOLINTNEXTLINE(google-runtime-int)
43 if (raw == static_cast<unsigned long>(-1) && (PyErr_Occurred() != nullptr))
44 {
45 PyErr_Clear();
46 return false;
47 }
48 if (raw > std::numeric_limits<std::uint32_t>::max())
49 {
50 return false;
51 }
52 value = sen::ObjectId(static_cast<std::uint32_t>(raw));
53 return true;
54 }
55
56 static handle cast(sen::ObjectId src, return_value_policy /* policy */, handle /* parent */)
57 {
58 return PyLong_FromUnsignedLong(src.get());
59 }
60};
